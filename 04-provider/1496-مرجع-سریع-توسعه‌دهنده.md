## 91\. Cheat Sheet

### بدون Parameter

```js
absoluteMethodHelper
    .runProvider(PROVIDER_ID)
```

### با Parameter

```js
absoluteMethodHelper
    .runProvider(
        PROVIDER_ID,
        [
            {
                name: 'Id',
                value: 1
            }
        ]
    )
```

### Structured Output

```js
.then(result => {
    const rows =
        JSON.parse(
            result.Content.providerResult || '[]'
        );
})
```

### File Input

```js
{
    name: 'file',
    value: myFile
}
```

### Blob → File

```js
const myFile =
    new File(
        [myBlob],
        'file.bin'
    );
```

### File Output

```js
.then(result => {
    const file = result.Content;

    saveAs(
        file,
        file.name
    );
})
```

### Error

```js
.catch(error => {
    console.error(error);
})
```

### Paging

```js
providerDesigner
    .runProviderPaging(
        PROVIDER_ID,
        parameters,
        pageIndex,
        pageSize,
        globalFilter,
        filters,
        orders
    )
```
